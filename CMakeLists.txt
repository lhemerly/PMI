cmake_minimum_required(VERSION 3.14)
project(pmi LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(PMI_WITH_LLAMA "Build the llama.cpp reference execution adapter" ON)
if(PMI_WITH_LLAMA)
  set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
  set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
  add_subdirectory(vendor/llama.cpp)
endif()
file(GLOB PMI_SOURCES CONFIGURE_DEPENDS src/*.cpp)
add_executable(pmi ${PMI_SOURCES})
target_include_directories(pmi PRIVATE include)
if(PMI_WITH_LLAMA)
  target_compile_definitions(pmi PRIVATE PMI_WITH_LLAMA=1)
  target_link_libraries(pmi PRIVATE llama)
  enable_testing()
  add_executable(pmi_type_sizes_test tests/type_sizes.cpp src/gguf.cpp)
  target_include_directories(pmi_type_sizes_test PRIVATE include)
  target_link_libraries(pmi_type_sizes_test PRIVATE ggml-base)
  add_test(NAME ggml_type_sizes COMMAND pmi_type_sizes_test)
endif()
